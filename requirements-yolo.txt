# Extra dependency for the standalone YOLO pick-place app (yolo_pickplace.py)
# on the Arduino VENTUNO Q. Install on top of the base requirements:
#   ~/miniforge3/envs/iotc-ventuno-xarm/bin/python3 -m pip install -r requirements-yolo.txt
#
# The NPU runtime on this board is TFLite + the QNN HTP delegate. The delegate
# (/usr/lib/libQnnTFLiteDelegate.so, apt package qairt-libs) and the models
# (/etc/models/yolox_quantized.tflite, midas_quantized.tflite) are installed by
# bootstrap-new-board.sh — the only Python dependency we add is the TFLite
# interpreter:

ai_edge_litert>=2.1.5

# Notes:
# - NOT onnxruntime: there is no aarch64-Linux QNN execution provider; stock
#   onnxruntime would run CPU-only and never touch the Hexagon NPU.
# - numpy + opencv come from the base requirements.txt / conda env.
